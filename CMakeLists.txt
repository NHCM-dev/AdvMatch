cmake_minimum_required(VERSION 3.20)
project(AdvMatch VERSION 0.0.1 LANGUAGES C CXX)

# Keep the IDE's compilation database explicit. GCC 16 defaults to C++20,
# but Clangd may otherwise infer an older language mode for headers.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# CLion's bundled Clangd can otherwise parse MinGW files as MSVC and miss
# GCC's private headers (for example bits/unique_ptr.h and mm_malloc.h).
if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpmachine
        OUTPUT_VARIABLE ADVMATCH_CLANGD_TARGET
        OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-file-name=include
        OUTPUT_VARIABLE ADVMATCH_CLANGD_GCC_INCLUDE
        OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(IS_DIRECTORY "${ADVMATCH_CLANGD_GCC_INCLUDE}")
        file(REAL_PATH "${ADVMATCH_CLANGD_GCC_INCLUDE}" ADVMATCH_CLANGD_GCC_INCLUDE)
        set(_advmatch_clangd_file "${CMAKE_CURRENT_SOURCE_DIR}/.clangd")
        if(EXISTS "${_advmatch_clangd_file}")
            file(STRINGS "${_advmatch_clangd_file}" _advmatch_clangd_first_line LIMIT_COUNT 1)
        endif()
        if(NOT EXISTS "${_advmatch_clangd_file}" OR
           _advmatch_clangd_first_line STREQUAL "# Generated by AdvMatch CMake for MinGW Clangd")
            configure_file(cmake/clangd-mingw.yaml.in "${_advmatch_clangd_file}" @ONLY)
        endif()
    endif()
endif()

include(FetchContent)
include(CTest)
include(GNUInstallDirs)

find_package(JNI QUIET)
option(ADVMATCH_BUILD_JNI "Build the JNI bridge" ${JNI_FOUND})

# Use the vendored headers without importing upstream build flags or tests.
find_package(Threads REQUIRED)
add_library(advmatch_disruptor INTERFACE)
add_library(AdvMatch::disruptor ALIAS advmatch_disruptor)
target_include_directories(advmatch_disruptor SYSTEM INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/disruptor-cpp/include")
target_link_libraries(advmatch_disruptor INTERFACE Threads::Threads)
target_compile_features(advmatch_disruptor INTERFACE cxx_std_20)

find_package(fmt 11.2 QUIET CONFIG)
if(NOT fmt_FOUND)
    set(FMT_INSTALL OFF CACHE BOOL "Do not install vendored fmt" FORCE)
    FetchContent_Declare(fmt
        GIT_REPOSITORY https://github.com/fmtlib/fmt.git
        GIT_TAG 40626af88bd7df9a5fb80be7b25ac85b122d6c21)
    FetchContent_MakeAvailable(fmt)
endif()

find_package(spdlog 1.15 QUIET CONFIG)
if(NOT spdlog_FOUND)
    set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "Use the selected fmt package" FORCE)
    FetchContent_Declare(spdlog
        GIT_REPOSITORY https://github.com/gabime/spdlog.git
        GIT_TAG 6fa36017cfd5731d617e1a934f0e5ea9c4445b13)
    FetchContent_MakeAvailable(spdlog)
endif()

add_library(advmatch_core STATIC
    src/advmatch/core.cpp
    include/advmatch/core.hpp
    src/advmatch/market/Asset.h
    src/advmatch/account/User.cpp
    src/advmatch/account/User.h
    src/advmatch/util/Utils.h
    src/advmatch/management/UserManager.cpp
    src/advmatch/management/UserManager.h
    src/advmatch/matching/MatchingEngine.cpp
    src/advmatch/matching/MatchingEngine.h
    src/advmatch/matching/Matching.h
        src/advmatch/matching/abstract/OrderBook.h
        src/advmatch/matching/orderbooks/SpotOrderBook.cpp
        src/advmatch/matching/orderbooks/SpotOrderBook.h
    src/advmatch/matching/QueuedMatchingEngine.cpp
    src/advmatch/matching/QueuedMatchingEngine.h
    src/advmatch/trading/Order.cpp
    src/advmatch/trading/Order.h
    src/advmatch/trading/OrderTypes.h
    src/advmatch/trading/Symbol.h
        src/advmatch/matching/abstract/RiskEngine.h
        src/advmatch/matching/riskengines/SpotRiskEngine.cpp
        src/advmatch/matching/riskengines/SpotRiskEngine.h
        src/advmatch/management/SymbolManager.cpp
        src/advmatch/management/SymbolManager.h)
add_library(AdvMatch::core ALIAS advmatch_core)
set_target_properties(advmatch_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(advmatch_core PUBLIC cxx_std_20)
target_compile_definitions(advmatch_core PRIVATE ADVMATCH_VERSION="${PROJECT_VERSION}")
target_include_directories(advmatch_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(advmatch_core PUBLIC AdvMatch::disruptor PRIVATE fmt::fmt spdlog::spdlog)

# Stable C symbols and plain C data types provide the boundary for JNI/PInvoke.
add_library(advmatch SHARED
    src/advmatch/c_api.cpp
    include/advmatch/advmatch.h
)
target_compile_features(advmatch PRIVATE cxx_std_20)
target_link_libraries(advmatch PRIVATE AdvMatch::core)
target_include_directories(advmatch PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(advmatch PROPERTIES OUTPUT_NAME advmatch)
if(WIN32)
    set_target_properties(advmatch PROPERTIES PREFIX "")
endif()

if(ADVMATCH_BUILD_JNI)
    find_package(JNI REQUIRED)
    add_library(advmatch_jni SHARED src/advmatch/jni_bridge.cpp)
    target_link_libraries(advmatch_jni PRIVATE advmatch JNI::JNI)
    set_target_properties(advmatch_jni PROPERTIES OUTPUT_NAME advmatch_jni)
    if(WIN32)
        set_target_properties(advmatch_jni PROPERTIES PREFIX "")
    endif()
endif()

if(BUILD_TESTING)
    add_executable(advmatch_smoke tests/smoke.cpp)
    target_compile_definitions(advmatch_smoke PRIVATE ADVMATCH_EXPECTED_VERSION="${PROJECT_VERSION}")
    target_link_libraries(advmatch_smoke PRIVATE AdvMatch::core advmatch)
    add_test(NAME advmatch_smoke COMMAND advmatch_smoke)

    add_executable(advmatch_matching_tests tests/matching.cpp)
    target_link_libraries(advmatch_matching_tests PRIVATE AdvMatch::core)
    add_test(NAME advmatch_matching_tests COMMAND advmatch_matching_tests)

    add_executable(advmatch_queued_matching_tests tests/queued_matching.cpp)
    target_link_libraries(advmatch_queued_matching_tests PRIVATE AdvMatch::core)
    add_test(NAME advmatch_queued_matching_tests COMMAND advmatch_queued_matching_tests)

    add_executable(advmatch_risk_tests tests/risk.cpp)
    target_link_libraries(advmatch_risk_tests PRIVATE AdvMatch::core)
    add_test(NAME advmatch_risk_tests COMMAND advmatch_risk_tests)
endif()

install(TARGETS advmatch LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/advmatch/advmatch.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/advmatch)
