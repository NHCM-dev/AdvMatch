cmake_minimum_required(VERSION 3.20)
project(AdvMatch VERSION 0.0.1 LANGUAGES C CXX)

if(POLICY CMP0167)
    cmake_policy(SET CMP0167 NEW)
endif()

# Keep the IDE's compilation database explicit. GCC 16 defaults to C++20,
# but Clangd may otherwise infer an older language mode for headers.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# CLion's bundled Clangd can otherwise parse MinGW files as MSVC and miss
# GCC's private headers (for example bits/unique_ptr.h and mm_malloc.h).
if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpmachine
        OUTPUT_VARIABLE ADVMATCH_CLANGD_TARGET
        OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-file-name=include
        OUTPUT_VARIABLE ADVMATCH_CLANGD_GCC_INCLUDE
        OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(IS_DIRECTORY "${ADVMATCH_CLANGD_GCC_INCLUDE}")
        file(REAL_PATH "${ADVMATCH_CLANGD_GCC_INCLUDE}" ADVMATCH_CLANGD_GCC_INCLUDE)
        set(_advmatch_clangd_file "${CMAKE_CURRENT_SOURCE_DIR}/.clangd")
        if(EXISTS "${_advmatch_clangd_file}")
            file(STRINGS "${_advmatch_clangd_file}" _advmatch_clangd_first_line LIMIT_COUNT 1)
        endif()
        if(NOT EXISTS "${_advmatch_clangd_file}" OR
           _advmatch_clangd_first_line STREQUAL "# Generated by AdvMatch CMake for MinGW Clangd")
            configure_file(cmake/clangd-mingw.yaml.in "${_advmatch_clangd_file}" @ONLY)
        endif()
    endif()
endif()

include(FetchContent)
include(CTest)
include(GNUInstallDirs)

find_package(JNI QUIET)
option(ADVMATCH_BUILD_JNI "Build the JNI bridge" ${JNI_FOUND})

# Use the vendored headers without importing upstream build flags or tests.
find_package(Threads REQUIRED)
add_library(advmatch_disruptor INTERFACE)
add_library(AdvMatch::disruptor ALIAS advmatch_disruptor)
target_include_directories(advmatch_disruptor SYSTEM INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/disruptor-cpp/include")
target_link_libraries(advmatch_disruptor INTERFACE Threads::Threads)
target_compile_features(advmatch_disruptor INTERFACE cxx_std_20)

find_package(fmt 11.2 QUIET CONFIG)
if(NOT fmt_FOUND)
    set(FMT_INSTALL OFF CACHE BOOL "Do not install vendored fmt" FORCE)
    FetchContent_Declare(fmt
        GIT_REPOSITORY https://github.com/fmtlib/fmt.git
        GIT_TAG 40626af88bd7df9a5fb80be7b25ac85b122d6c21)
    FetchContent_MakeAvailable(fmt)
endif()

find_package(spdlog 1.15 QUIET CONFIG)
if(NOT spdlog_FOUND)
    set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "Use the selected fmt package" FORCE)
    FetchContent_Declare(spdlog
        GIT_REPOSITORY https://github.com/gabime/spdlog.git
        GIT_TAG 6fa36017cfd5731d617e1a934f0e5ea9c4445b13)
    FetchContent_MakeAvailable(spdlog)
endif()

find_package(Boost 1.92 QUIET)
add_library(advmatch_boost_headers INTERFACE)
add_library(AdvMatch::boost_headers ALIAS advmatch_boost_headers)
if(Boost_FOUND)
    target_include_directories(advmatch_boost_headers SYSTEM INTERFACE ${Boost_INCLUDE_DIRS})
else()
    FetchContent_Declare(advmatch_boost
        URL https://archives.boost.io/release/1.92.0/source/boost_1_92_0.tar.gz
        URL_HASH SHA256=c4a3b310ddd2472416e091067166b0713be97c63f38c212c484ada022fd296ce
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE
        SOURCE_SUBDIR __advmatch_headers_only)
    FetchContent_MakeAvailable(advmatch_boost)
    target_include_directories(advmatch_boost_headers SYSTEM INTERFACE ${advmatch_boost_SOURCE_DIR})
endif()

find_package(unordered_dense 5.0.1 QUIET CONFIG)
if(NOT unordered_dense_FOUND)
    FetchContent_Declare(unordered_dense
        GIT_REPOSITORY https://github.com/martinus/unordered_dense.git
        GIT_TAG 1c8636e810d7e4a485d17bb9cbd6dccb5fe4e284)
    FetchContent_MakeAvailable(unordered_dense)
endif()

find_package(absl 20250814 QUIET CONFIG)
if(NOT absl_FOUND)
    set(ABSL_BUILD_TESTING OFF CACHE BOOL "Do not build Abseil tests" FORCE)
    set(ABSL_ENABLE_INSTALL OFF CACHE BOOL "Do not install vendored Abseil" FORCE)
    set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "Use the parent C++ standard" FORCE)
    if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.28)
        set(ADVMATCH_ABSL_EXCLUDE_FROM_ALL EXCLUDE_FROM_ALL)
    endif()
    FetchContent_Declare(absl
        GIT_REPOSITORY https://github.com/abseil/abseil-cpp.git
        GIT_TAG 0cf0a5c9d12cc3783363ab20f11613e69fd04c9a
        ${ADVMATCH_ABSL_EXCLUDE_FROM_ALL})
    FetchContent_MakeAvailable(absl)
endif()

    PUBLIC AdvMatch::disruptor AdvMatch::boost_headers unordered_dense::unordered_dense absl::flat_hash_map absl::node_hash_map
    PRIVATE fmt::fmt spdlog::spdlog)

# Stable C symbols and plain C data types provide the boundary for JNI/PInvoke.
add_library(advmatch SHARED
    src/advmatch/c_api.cpp
    include/advmatch/advmatch.h
)
target_compile_features(advmatch PRIVATE cxx_std_20)
target_link_libraries(advmatch PRIVATE AdvMatch::core)
target_include_directories(advmatch PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(advmatch PROPERTIES OUTPUT_NAME advmatch)
if(WIN32)
    set_target_properties(advmatch PROPERTIES PREFIX "")
endif()
